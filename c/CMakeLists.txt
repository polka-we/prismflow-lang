cmake_minimum_required(VERSION 3.10)
project(prism C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Оптимизация включена всегда, в том числе в debug-сборке Android (иначе C-код идёт с -O0 и в 2,5-3 раза медленнее).
add_library(prism_core STATIC src/prism.c)
if(MSVC)
  target_compile_options(prism_core PRIVATE /O2)
else()
  target_compile_options(prism_core PRIVATE -O2 -fno-stack-protector)
endif()
target_include_directories(prism_core PUBLIC include)
find_library(M_LIB m)
if(M_LIB)
  target_link_libraries(prism_core PUBLIC ${M_LIB})
endif()

if(ANDROID)
  # Android NDK: общая библиотека libprism.so с JNI-обвязкой
  add_library(prism SHARED android/jni/prism_jni.c)
  target_link_libraries(prism prism_core log)
else()
  add_executable(prism_cli src/main.c)
  set_target_properties(prism_cli PROPERTIES OUTPUT_NAME prism)
  target_link_libraries(prism_cli prism_core)
  find_package(Threads)
  if(Threads_FOUND)
    target_link_libraries(prism_cli Threads::Threads)
  endif()
endif()
